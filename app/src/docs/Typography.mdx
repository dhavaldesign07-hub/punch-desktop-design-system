export function TypeRow({ name, token, size, lineHeight, weight }) {
  const weightMap = { 700: 'Bold', 600: 'Semi Bold', 500: 'Medium', 400: 'Regular' };
  return (
    <div style={{
      display: 'flex',
      alignItems: 'baseline',
      gap: 24,
      padding: '12px 0',
      borderBottom: '1px solid #2a2e40',
    }}>
      <div style={{ minWidth: 180, flexShrink: 0 }}>
        <strong style={{ fontSize: 13 }}>{name}</strong>
        <div style={{ fontSize: 11, color: '#888', fontFamily: 'monospace' }}>
          {size} / {lineHeight} / {weightMap[weight] || weight}
        </div>
        <code style={{ fontSize: 10, color: '#666' }}>{token}</code>
      </div>
      <p style={{
        fontFamily: "'Inter', system-ui, sans-serif",
        fontSize: size,
        lineHeight: lineHeight,
        fontWeight: weight,
        margin: 0,
        whiteSpace: 'nowrap',
        overflow: 'hidden',
        textOverflow: 'ellipsis',
      }}>
        The quick brown fox jumps over the lazy dog
      </p>
    </div>
  );
}

export function TypeSection({ title, description, children }) {
  return (
    <div style={{ marginBottom: 32 }}>
      <h3 style={{ marginBottom: 4 }}>{title}</h3>
      {description && <p style={{ fontSize: 13, color: '#888', marginTop: 0 }}>{description}</p>}
      {children}
    </div>
  );
}

## Token Reference

Each text style has three tokens following the pattern `--font-{category}-{size}-{weight}-{property}`:

| Token suffix | Property |
|---|---|
| `-size` | `font-size` |
| `-line-height` | `line-height` |
| `-weight` | `font-weight` |

**Example usage:**

```css
.heading {
  font-family: var(--font-family-base);
  font-size: var(--font-heading-h1-bold-size);
  line-height: var(--font-heading-h1-bold-line-height);
  font-weight: var(--font-heading-h1-bold-weight);
}
```
